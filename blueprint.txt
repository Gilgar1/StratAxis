3. DATABASE SCHEMA AND DATA VERSIONING

3.1. MongoDB Collections Schema

3.1.1. User Collection
    {
        _id: ObjectId,
        email: String (unique, indexed, required),
        password: String (hashed, required),
        role: String (enum: ['FREE_USER', 'PAID_USER', 'ADMIN'], default: 'FREE_USER', indexed),
        firstName: String,
        lastName: String,
        phone: String,
        createdAt: Date (indexed),
        updatedAt: Date,
        lastLogin: Date,
        isActive: Boolean (default: true),
        resetPasswordToken: String,
        resetPasswordExpires: Date,
        subscriptionExpires: Date (for PAID_USER)
    }

3.1.2. Property Collection
    {
        _id: ObjectId,
        title: String (required, indexed for text search),
        description: String,
        city: String (enum: ['Yaoundé', 'Douala'], required, indexed),
        neighborhood: String (indexed),
        location: {
            type: String (enum: ['Point'], default: 'Point'),
            coordinates: [Number] (longitude, latitude, indexed geospatial)
        },
        propertyType: String (enum: ['apartment', 'house', 'land', 'commercial'], required, indexed),
        price: Number (required, > 0),
        currency: String (default: 'XAF'),
        size: Number (m², required, > 0),
        pricePerM2: Number (computed, indexed),
        bedrooms: Number,
        bathrooms: Number,
        images: [String] (URLs),
        dataSource: ObjectId (reference to DataSource, indexed),
        dataSourceRecordId: String (original ID from source),
        qualityScore: Number (0-100, validation score),
        validationStatus: String (enum: ['pending', 'validated', 'rejected'], default: 'pending'),
        createdAt: Date (indexed),
        updatedAt: Date,
        scrapedAt: Date (when data was collected),
        version: Number (default: 1)
    }

3.1.3. Listing Collection (aggregated/processed data)
    {
        _id: ObjectId,
        city: String (required, indexed),
        neighborhood: String (indexed),
        propertyType: String (required, indexed),
        period: String (enum: ['daily', 'weekly', 'monthly'], default: 'monthly'),
        periodStart: Date (indexed),
        periodEnd: Date,
        avgPrice: Number,
        medianPrice: Number,
        minPrice: Number,
        maxPrice: Number,
        avgPricePerM2: Number,
        propertyCount: Number,
        trendDirection: String (enum: ['up', 'down', 'stable']),
        trendPercentage: Number,
        createdAt: Date
    }

3.1.4. Booking Collection
    {
        _id: ObjectId,
        user: ObjectId (reference to User, required, indexed),
        consultationType: String (enum: ['market_analysis', 'investment_advice', 'property_valuation'], required),
        preferredDate: Date (required),
        preferredTime: String,
        status: String (enum: ['pending', 'confirmed', 'completed', 'cancelled'], default: 'pending', indexed),
        notes: String,
        adminNotes: String,
        createdAt: Date (indexed),
        updatedAt: Date,
        confirmedAt: Date,
        completedAt: Date
    }

3.1.5. DataSource Collection
    {
        _id: ObjectId,
        name: String (required, unique),
        type: String (enum: ['scraper', 'ocr', 'manual'], required),
        sourceUrl: String (for scrapers),
        sourcePath: String (for OCR, file path),
        isActive: Boolean (default: true),
        lastRunAt: Date,
        lastRunStatus: String (enum: ['success', 'failed', 'partial']),
        recordsCollected: Number,
        recordsValidated: Number,
        recordsRejected: Number,
        config: Object (source-specific configuration),
        createdAt: Date,
        updatedAt: Date
    }

3.1.6. MLModel Collection
    {
        _id: ObjectId,
        name: String (required, e.g., 'price_prediction_v1'),
        version: String (required, semantic versioning),
        type: String (enum: ['price_prediction', 'trend_forecast'], required),
        algorithm: String (e.g., 'random_forest', 'prophet', 'xgboost'),
        status: String (enum: ['training', 'active', 'archived'], default: 'training', indexed),
        metrics: {
            mse: Number,
            mae: Number,
            r2: Number,
            rmse: Number
        },
        featureImportance: Object (feature names and importance scores),
        trainingDataRange: {
            startDate: Date,
            endDate: Date,
            recordCount: Number
        },
        trainedAt: Date,
        deployedAt: Date,
        modelPath: String (file path to serialized model),
        config: Object (hyperparameters, feature list)
    }

3.1.7. PricePrediction Collection (prediction cache/history)
    {
        _id: ObjectId,
        model: ObjectId (reference to MLModel),
        input: {
            city: String,
            propertyType: String,
            size: Number,
            neighborhood: String (optional)
        },
        prediction: Number (predicted price),
        confidenceInterval: {

            upper: Number

3.2. Indexes

3.2.1. Property Collection Indexes
    1. city, propertyType (compound index)
    2. price (ascending)
    3. pricePerM2 (ascending)
    4. createdAt (descending)
    5. location (2dsphere, geospatial)
    6. title (text index for search)
    7. city, neighborhood (compound index)
    8. dataSource, validationStatus (compound index)

3.2.2. Listing Collection Indexes
    1. city, propertyType, periodStart (compound index)
    2. city, neighborhood, periodStart (compound index)
    3. periodStart (descending)

3.2.3. User Collection Indexes
    1. email (unique)
    2. role (for RBAC queries)
    3. createdAt (for user analytics)

3.2.4. Booking Collection Indexes
    1. user (for user's bookings)
    2. status, preferredDate (for admin queries)
    3. preferredDate (for availability checks)

3.3. Data Versioning Strategy

3.3.1. Property Versioning
    1. New properties: version = 1
    2. Updates from re-scraping: increment version, keep previous version in history collection
    3. History collection: PropertyHistory (same schema + version field, no indexes on version)
    4. Current properties: only latest version (version field in Property collection for tracking)
    5. Historical analysis: query PropertyHistory for trend analysis

3.3.2. ML Model Versioning
    1. Semantic versioning: major.minor.patch (e.g., 1.2.3)
    2. Major: breaking changes to API or significant algorithm change
    3. Minor: new features, improved accuracy
    4. Patch: bug fixes, parameter tuning
    5. Only one active model per type (price_prediction, trend_forecast)
    6. Archive previous models (status = 'archived'), keep for comparison

3.3.3. Listing Aggregation Versioning
    1. Listings are time-period snapshots (not versioned)
    2. Recalculate monthly (previous month's data becomes historical)
    3. Keep last 24 months of listing aggregates for trend analysis
    4. Delete older aggregates (or archive to cold storage if needed)

4. DATA ENGINEERING PIPELINE STEPS

4.1. Scraping Pipeline
    1. Scheduler triggers scraper (cron: daily at 02:00 UTC)
    2. Scraper factory selects active scrapers from DataSource collection
    3. For each scraper:
       a. Load scraper configuration (base URL, selectors, rate limits)
       b. Fetch listing pages (respect robots.txt, delays between requests)
       c. Parse HTML with BeautifulSoup/Selenium
       d. Extract property data (title, price, location, type, size, bedrooms, images)
       e. Normalize data format (common schema)
       f. Log extraction results (records found, errors)
    4. Save raw scraped data to temporary JSON files
    5. Return extracted records to ETL pipeline

4.2. OCR Pipeline
    1. Scheduler triggers OCR processor (cron: weekly, Sunday 03:00 UTC)
    2. Load active OCR sources from DataSource collection
    3. For each PDF source:
       a. Load PDF file from configured path
       b. Convert PDF pages to images (pdf2image)
       c. Preprocess images (contrast, noise reduction, deskew)
       d. Run Tesseract OCR on each page
       e. Extract text from OCR output
       f. Parse structured data using regex patterns (price, location, date, property details)
       g. Normalize extracted data (currency, dates, locations)
       h. Log extraction results (pages processed, data extracted, errors)
    4. Save parsed OCR data to temporary JSON files
    5. Return extracted records to ETL pipeline

4.3. ETL Pipeline Orchestration
    1. Pipeline orchestrator (pipeline.py) receives data from scrapers/OCR
    2. Extract phase:
       a. Load data from scrapers (JSON files)
       b. Load data from OCR (JSON files)
       c. Merge all extracted records
       d. Tag each record with source type and ID
    3. Transform phase:
       a. Currency normalization (XAF to standard, handle currency symbols)
       b. Location standardization (neighborhood name mapping, city validation)
       c. Property type categorization (map variations to standard types)
       d. Unit conversion (acres/feet to m², handle missing units)
       e. Date parsing (multiple formats → ISO 8601)
       f. Price per m² calculation (price / size)
       g. Geocoding (location names → coordinates, API or static mapping)
       h. Data enrichment (add metadata: source, extraction date)
    4. Validation phase (see section 4.4)
    5. Load phase:
       a. Check for duplicates (compare dataSource + dataSourceRecordId)
       b. If duplicate: compare versions, update if newer or higher quality
       c. If new: insert into Property collection
       d. Update DataSource collection (records collected, validated, rejected)
       e. Log pipeline execution metrics
    6. Post-load:
       a. Trigger listing aggregation recalculation
       b. Trigger ML model retraining flag (if significant new data)

4.4. Validation Pipeline
    1. Schema validation (schema_validator.py):
       a. Check required fields present
       b. Validate data types (price is number, size is number, dates are dates)
       c. Validate ranges (price > 0, size > 0, coordinates within Cameroon bounds)
       d. Validate enums (city in ['Yaoundé', 'Douala'], propertyType in allowed list)
       e. Reject if schema validation fails
    2. Quality scoring (data_quality.py):
       a. Completeness: count non-null fields / total fields (weight: 30%)
       b. Consistency: check logical consistency (weight: 40%)
          - Price matches property type (apartments < 100M XAF in target cities)
          - Size matches property type (apartments typically 30-200 m²)
          - Location matches city (neighborhood belongs to city)
          - Price per m² within reasonable range (1,000 - 500,000 XAF/m²)
       c. Accuracy indicators (weight: 30%):
          - OCR confidence score (if from OCR source)
          - Scraper extraction quality (if from scraper, check field completeness)
       d. Calculate quality score: (completeness * 0.3) + (consistency * 0.4) + (accuracy * 0.3)
       e. Reject if quality score < 70%
    3. Outlier detection (outlier_detection.py):
       a. Statistical outliers: Z-score > 3 for price per m² (by city, property type)
       b. Business logic outliers:
          - Price per m² < 1,000 XAF (likely data error)
          - Price per m² > 500,000 XAF (likely luxury/commercial, flag for review)
          - Size > 1,000 m² for apartments (likely house mislabeled)
          - Price > 1,000,000,000 XAF (likely data entry error)
       c. Flag outliers (validationStatus = 'pending', qualityScore reduced by 10 points)
       d. Do not auto-reject outliers (require manual review)
    4. Validation result:
       a. validationStatus: 'validated' (qualityScore >= 70%, no critical outliers)
       b. validationStatus: 'pending' (qualityScore >= 70%, but outliers detected)
       c. validationStatus: 'rejected' (qualityScore < 70%)
    5. Generate validation report:
       a. Total records processed
       b. Validated count
       c. Pending count (requires review)
       d. Rejected count
       e. Average quality score
       f. Common rejection reasons
       g. Top outlier properties (for review)

5. MACHINE LEARNING WORKFLOW

5.1. Feature Engineering

5.1.1. Data Preparation
    1. Load validated properties from MongoDB (validationStatus = 'validated')
    2. Filter by date range (last 24 months minimum, all historical if available)
    3. Filter by cities: Yaoundé, Douala only
    4. Remove properties with missing critical features (price, size, propertyType, city)
    5. Handle missing values:
       a. bedrooms: impute with median by propertyType and city
       b. bathrooms: impute with bedrooms * 0.75 (typical ratio)
       c. neighborhood: use 'unknown' category
       d. location coordinates: geocode neighborhood center if missing

5.1.2. Feature Extraction
    1. Numerical features:
       a. size (m²)
       b. bedrooms
       c. bathrooms
       d. pricePerM2 (target variable for price prediction)
       e. age (years since scrapedAt if available, else 0)
    2. Categorical features (one-hot encoded):
       a. city (Yaoundé, Douala)
       b. propertyType (apartment, house, land, commercial)
       c. neighborhood (top 20 by frequency, rest as 'other')
    3. Temporal features:
       a. year (extracted from scrapedAt)
       b. month (1-12, cyclical encoding: sin/cos)
       c. quarter (1-4, cyclical encoding)
    4. Derived features:
       a. size_category (small: < 50m², medium: 50-150m², large: > 150m²)
       b. bedrooms_per_m2 (bedrooms / size)
       c. neighborhood_price_avg (average price per m² in neighborhood, grouped)
       d. city_property_type_avg (average price per m² by city and property type)
    5. Geospatial features (if coordinates available):
       a. distance_to_city_center (km, approximate centers: Yaoundé 3.87°N 11.52°E, Douala 4.05°N 9.70°E)
       b. latitude
       c. longitude

5.1.3. Feature Selection
    1. Remove features with > 50% missing values
    2. Remove low-variance features (variance < 0.01)
    3. Remove highly correlated features (correlation > 0.95, keep one)
    4. Use feature importance from Random Forest (initial model) to rank features
    5. Select top 15-20 features for final model (balance accuracy vs complexity)

5.2. Model Selection

5.2.1. Price Prediction Model Candidates
    1. Random Forest Regressor (baseline, handles non-linearity, feature importance)
    2. XGBoost Regressor (gradient boosting, good for tabular data)
    3. Linear Regression with regularization (Lasso/Ridge, interpretable baseline)
    4. Decision Tree Regressor (simple baseline)
    5. Evaluation criteria:
       a. Primary: RMSE (root mean squared error), MAE (mean absolute error)
       b. Secondary: R² (coefficient of determination)
       c. Business metric: percentage of predictions within 20% of actual price

5.2.2. Trend Forecasting Model Candidates
    1. Prophet (Facebook, handles seasonality, holidays, trend changes)
    2. ARIMA (AutoRegressive Integrated Moving Average, classical time series)
    3. Exponential Smoothing (simple baseline)
    4. Evaluation criteria:
       a. Primary: RMSE, MAE on test set (last 3-6 months)
       b. Secondary: MAPE (mean absolute percentage error)
       c. Visual inspection of forecast vs actual (trend direction accuracy)

5.2.3. Model Selection Process
    1. Split data: 70% train, 15% validation, 15% test
    2. Train all candidate models on training set
    3. Evaluate on validation set
    4. Select best model based on evaluation criteria
    5. Final evaluation on held-out test set
    6. Document model selection rationale (metrics, trade-offs)

5.3. Training

5.3.1. Price Prediction Model Training
    1. Load preprocessed features (from feature engineering)
    2. Split into train/validation/test (70/15/15)
    3. Hyperparameter tuning (GridSearchCV or RandomizedSearchCV):
       a. Random Forest: n_estimators (50-500), max_depth (5-20), min_samples_split (2-10)
       b. XGBoost: n_estimators (50-500), max_depth (3-10), learning_rate (0.01-0.3)
    4. Train best model with optimal hyperparameters on training set
    5. Evaluate on validation set, adjust if needed
    6. Final training on train + validation combined (after validation confirms good performance)
    7. Save model artifact (joblib or pickle)
    8. Calculate feature importance
    9. Store model metadata in MLModel collection:
       a. Algorithm name
       b. Hyperparameters
       c. Feature list
       d. Training date range
       e. Training record count
       f. Metrics (MSE, MAE, RMSE, R²)

5.3.2. Trend Forecasting Model Training
    1. Aggregate data by time period (monthly averages by city and property type)
    2. Prepare time series data (date, average_price_per_m2)
    3. Split into train/test (last 3 months as test set)
    4. Train Prophet model:
       a. Configure seasonality (yearly, monthly)
       b. Set changepoint_prior_scale (0.05-0.5, controls trend flexibility)
       c. Add custom seasonalities if needed (quarterly, semi-annual)
    5. Generate forecasts for test period
    6. Evaluate forecast accuracy (RMSE, MAE, MAPE)
    7. If satisfactory, train final model on all data
    8. Save model artifact
    9. Store model metadata in MLModel collection

5.4. Evaluation

5.4.1. Price Prediction Evaluation
    1. Test set evaluation:
       a. Generate predictions for all test set properties
       b. Calculate RMSE, MAE, R²
       c. Calculate percentage within 10%, 20%, 30% of actual price
       d. Analyze errors by property type, city, price range
    2. Business metrics:
       a. Median absolute percentage error (MAPE)
       b. Prediction accuracy by city (Yaoundé vs Douala)
       c. Prediction accuracy by property type
       d. Prediction accuracy by price range (low, medium, high)
    3. Feature importance analysis:
       a. Extract feature importance from trained model
       b. Rank features by importance
       c. Document which features contribute most to predictions
    4. Error analysis:
       a. Identify properties with highest prediction errors
       b. Analyze common characteristics of poorly predicted properties
       c. Document insights for future feature engineering

5.4.2. Trend Forecasting Evaluation
    1. Test set evaluation:
       a. Generate forecasts for last 3-6 months
       b. Calculate RMSE, MAE, MAPE
       c. Check if forecasted trend direction matches actual (up/down/stable)
       d. Calculate trend direction accuracy percentage
    2. Visual evaluation:
       a. Plot forecast vs actual time series
       b. Plot confidence intervals
       c. Identify periods with large forecast errors
    3. Business metrics:
       a. Forecast accuracy by city
       b. Forecast accuracy by property type
       c. Forecast accuracy by forecast horizon (1 month, 3 months, 6 months)

5.5. Deployment

5.5.1. Model Deployment Pipeline
    1. Model validation threshold:
       a. Price prediction: R² > 0.6, RMSE < 20% of mean price
       b. Trend forecast: MAPE < 15%, trend direction accuracy > 70%
       c. If thresholds not met: retrain with more data or different features
    2. Model promotion:
       a. Archive current active model (status = 'archived')
       b. Set new model as active (status = 'active')
       c. Update MLModel collection
       d. Move model artifact to production path
    3. API integration:
       a. Load model artifact in prediction service
       b. Implement prediction function (input → feature engineering → model.predict → output)
       c. Add caching layer (cache predictions for identical inputs, 24h TTL)
       d. Add monitoring (log prediction requests, response times, errors)

5.5.2. Model Retraining Strategy
    1. Trigger conditions:
       a. Scheduled: monthly retraining with new data
       b. On-demand: admin triggers via API
       c. Automatic: if new validated properties > 1000 since last training
    2. Retraining process:
       a. Load all validated properties (updated dataset)
       b. Re-run feature engineering
       c. Retrain model with same hyperparameters (or re-tune if performance degraded)
       d. Evaluate new model vs current active model
       e. If new model performs better: deploy (A/B testing not required for MVP)
       f. If performance similar or worse: keep current model, log for review

8. SECURITY CONSIDERATIONS

8.1. Authentication Security
    1. Password requirements: minimum 8 characters, at least one letter and one number
    2. Password hashing: bcrypt with salt rounds = 10
    3. JWT tokens: 
       - Access token: 15 minutes expiry
       - Refresh token: 7 days expiry, stored in httpOnly cookie (if possible) or secure storage
       - Token signing: use strong secret (environment variable, min 32 characters)
    4. Password reset: 
       - Generate secure random token (crypto.randomBytes)
       - Token expiry: 1 hour
       - Store hashed token in database
       - Send reset link via email (stub email service for MVP)
    5. Account lockout: after 5 failed login attempts, lock account for 30 minutes (implement in middleware)

8.2. API Security
    1. Input validation: validate all request parameters and body fields
    2. SQL injection prevention: not applicable (MongoDB with parameterized queries)
    3. NoSQL injection prevention: sanitize user inputs, use Mongoose schema validation
    4. XSS prevention: sanitize user inputs, use helmet.js middleware
    5. CSRF protection: use CSRF tokens for state-changing operations (POST, PUT, DELETE)
    6. Rate limiting:
       - Public endpoints: 100 requests/minute per IP
       - Authenticated endpoints: 200 requests/minute per user
       - Prediction endpoints: 100 requests/hour per user
       - Admin endpoints: 500 requests/minute per admin
    7. CORS: configure allowed origins (frontend domain only in production)
    8. API versioning: /api/v1/ prefix for future compatibility

8.3. Data Protection
    1. Sensitive data encryption:
       - Passwords: hashed (bcrypt)
       - JWT tokens: signed, not encrypted (no sensitive data in payload)
       - Database connection: use TLS/SSL for MongoDB connection
    2. Environment variables: never commit .env files, use .env.example template
    3. API keys: store in environment variables, never in code
    4. User data: 
       - Do not expose password hashes in API responses
       - Limit user data returned to necessary fields only
       - Implement data access controls (users can only access their own data)
    5. Logging: do not log sensitive data (passwords, tokens, credit card numbers)

8.4. Scraping Ethics and Legal
    1. robots.txt compliance: check and respect robots.txt before scraping
    2. Rate limiting: implement delays between requests (1-2 seconds minimum)
    3. User-Agent: identify scraper in User-Agent header ("StratAxis-Bot/1.0")
    4. Terms of Service: review and comply with target website terms
    5. Data usage: only use scraped data for analytics, not republish full listings
    6. Attribution: acknowledge data sources in documentation (if required)
    7. Contact information: provide contact email in User-Agent for website owners
    8. Government PDFs: public domain or fair use, but verify local regulations
    9. Data retention: follow data retention policies (delete outdated data)

8.5. Infrastructure Security
    1. HTTPS: enable SSL/TLS certificates (Let's Encrypt free certificates)
    2. Server hardening: 
       - Disable unnecessary services
       - Keep OS and software updated
       - Use firewall (UFW or iptables)
       - Allow only necessary ports (80, 443, 22 for SSH)
    3. Database security:
       - Use MongoDB authentication (username/password)
       - Restrict database access to application server IP only
       - Use MongoDB Atlas network whitelist (if using Atlas)
    4. Environment separation: separate development, staging, production environments
    5. Secrets management: use environment variables, consider secret management service for production

8.6. RBAC Security
    1. Role validation: check user role on every protected endpoint
    2. Permission checks: implement fine-grained permissions within roles
    3. Admin actions audit: log all admin actions (user role changes, data source changes, model deployments)
    4. Principle of least privilege: users get minimum required permissions
    5. Role escalation prevention: prevent users from modifying their own roles

9. DEPLOYMENT STEPS

9.1. Pre-Deployment Preparation
    1. Choose hosting provider (VPS: DigitalOcean, Linode, AWS EC2, or equivalent)
    2. Select server specifications: minimum 2GB RAM, 1 CPU, 20GB SSD (scalable)
    3. Choose domain name and configure DNS
    4. Set up MongoDB: MongoDB Atlas (cloud) or self-hosted on VPS
    5. Prepare environment variables for production (.env files)
    6. Set up GitHub repository (private or public, as required)

9.2. Server Setup
    1. Initialize VPS: Ubuntu 20.04 LTS or 22.04 LTS
    2. Create non-root user with sudo privileges
    3. Update system packages: apt update && apt upgrade
    4. Install Node.js (v18 LTS): use NodeSource repository
    5. Install Python 3.9+: apt install python3 python3-pip
    6. Install MongoDB (if self-hosting): use MongoDB official repository
    7. Install Nginx: apt install nginx
    8. Install PM2 (Node.js process manager): npm install -g pm2
    9. Configure firewall: ufw allow 22 (SSH), 80 (HTTP), 443 (HTTPS), ufw enable
    10. Set up SSH key authentication (disable password authentication)

9.3. SSL Certificate Setup
    1. Install Certbot: apt install certbot python3-certbot-nginx
    2. Configure Nginx for initial HTTP setup
    3. Obtain SSL certificate: certbot --nginx -d yourdomain.com -d www.yourdomain.com
    4. Configure automatic renewal: certbot renew --dry-run (test)
    5. Set up cron job for certificate renewal: 0 0 * * * certbot renew --quiet

9.4. MongoDB Setup
    9.4.1. MongoDB Atlas (Cloud - Recommended)
        1. Create MongoDB Atlas account
        2. Create cluster (free tier: M0, or paid for production)
        3. Configure network access (whitelist VPS IP)
        4. Create database user (admin user, strong password)
        5. Get connection string
        6. Update backend .env with connection string

    9.4.2. Self-Hosted MongoDB
        1. Install MongoDB on VPS
        2. Configure MongoDB authentication (create admin user)
        3. Configure MongoDB bind address (127.0.0.1 or specific IP)
        4. Enable MongoDB service: systemctl enable mongod, systemctl start mongod
        5. Create application database and user
        6. Update backend .env with connection string

9.5. Backend Deployment
    1. Clone repository on VPS: git clone <repo-url>
    2. Navigate to backend directory: cd backend
    3. Install dependencies: npm install --production
    4. Create .env file with production environment variables:
       - NODE_ENV=production
       - PORT=3000
       - MONGODB_URI=<connection-string>
       - JWT_SECRET=<strong-secret>
       - JWT_EXPIRES_IN=15m
       - JWT_REFRESH_EXPIRES_IN=7d
       - FRONTEND_URL=<frontend-url>
    5. Build application (if TypeScript): npm run build
    6. Start application with PM2: pm2 start server.js --name strataxis-backend
    7. Configure PM2 to start on boot: pm2 startup, pm2 save
    8. Set up PM2 monitoring: pm2 monit
    9. Test backend health endpoint: curl http://localhost:3000/api/health

9.6. Data Pipeline Deployment
    1. Navigate to data-pipeline directory: cd data-pipeline
    2. Create Python virtual environment: python3 -m venv venv
    3. Activate virtual environment: source venv/bin/activate
    4. Install dependencies: pip install -r requirements.txt
    5. Create config.yaml with production settings (MongoDB URI, paths)
    6. Set up cron jobs for scheduled execution:
       - Scraping: 0 2 * * * (daily at 02:00 UTC)
       - OCR: 0 3 * * 0 (weekly, Sunday at 03:00 UTC)
    7. Test pipeline manually: python src/main.py
    8. If using Python ML service: deploy similar to backend (separate PM2 process or systemd service)

9.7. Frontend Deployment
    1. Build React application: npm run build (creates build/ directory)
    2. Copy build/ directory to server: scp -r build/ user@server:/var/www/strataxis-frontend
    3. On server, configure Nginx to serve static files:
       - Location: /var/www/strataxis-frontend
       - Nginx configuration: proxy API requests to backend, serve static files for frontend routes
    4. Configure Nginx reverse proxy for API:
       - /api/* → proxy to http://localhost:3000
       - / → serve React build/index.html (SPA routing)
    5. Test frontend: access https://yourdomain.com

9.8. Nginx Configuration
    1. Create Nginx config file: /etc/nginx/sites-available/strataxis
    2. Configuration structure:
       - Server block for HTTP (redirect to HTTPS)
       - Server block for HTTPS (SSL certificates, proxy to backend, serve frontend)
    3. Enable site: ln -s /etc/nginx/sites-available/strataxis /etc/nginx/sites-enabled/
    4. Test Nginx configuration: nginx -t
    5. Reload Nginx: systemctl reload nginx

9.9. Post-Deployment Verification
    1. Test all API endpoints (health check, auth, properties, analytics)
    2. Test frontend pages (Home, Login, Register, Dashboard, Analytics)
    3. Test authentication flow (register, login, protected routes)

    2000: AUTHENTICATION_ERROR
    2001: INVALID_CREDENTIALS
    2002: TOKEN_EXPIRED
    2003: TOKEN_INVALID
    2004: ACCOUNT_LOCKED
    2005: ACCOUNT_INACTIVE
    3000: AUTHORIZATION_ERROR
    3001: INSUFFICIENT_PERMISSIONS
    3002: ROLE_REQUIRED
    4000: NOT_FOUND
    4001: USER_NOT_FOUND
    4002: PROPERTY_NOT_FOUND
    4003: BOOKING_NOT_FOUND
    5000: SERVER_ERROR
    5001: DATABASE_ERROR
    5002: EXTERNAL_SERVICE_ERROR
    5003: ML_SERVICE_ERROR
    5004: DATA_PIPELINE_ERROR
    6000: RATE_LIMIT_EXCEEDED
    6001: TOO_MANY_REQUESTS

15.3.2. HTTP Status Code Mapping
    200: Success
    201: Created
    400: Bad Request (validation errors, error codes 1000-1003)
    401: Unauthorized (authentication errors, error codes 2000-2005)
    403: Forbidden (authorization errors, error codes 3000-3002)
    404: Not Found (resource not found, error codes 4000-4003)
    409: Conflict (duplicate resource, e.g., email exists)
    429: Too Many Requests (rate limit exceeded, error codes 6000-6001)
    500: Internal Server Error (server errors, error codes 5000-5004)
    503: Service Unavailable (ML service unavailable, pipeline unavailable)

    # HTTP server - redirect to HTTPS

        listen 80;
        listen [::]:80;
        server_name yourdomain.com www.yourdomain.com;
        return 301 https://$server_name$request_uri;
    }
    
    # HTTPS server
    server {
        listen 443 ssl http2;
        listen [::]:443 ssl http2;
        server_name yourdomain.com www.yourdomain.com;
        
        # SSL certificates (Let's Encrypt)
        ssl_certificate /etc/letsencrypt/live/yourdomain.com/fullchain.pem;
        ssl_certificate_key /etc/letsencrypt/live/yourdomain.com/privkey.pem;
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers HIGH:!aNULL:!MD5;
        ssl_prefer_server_ciphers on;
        
        # Security headers
        add_header X-Frame-Options "SAMEORIGIN" always;
        add_header X-Content-Type-Options "nosniff" always;
        add_header X-XSS-Protection "1; mode=block" always;
        add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;
        
        # Frontend static files
        root /var/www/strataxis-frontend;
        index index.html;
        
        # API proxy to backend
        location /api {
            proxy_pass http://localhost:3000;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection 'upgrade';
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_cache_bypass $http_upgrade;
            proxy_read_timeout 300s;
            proxy_connect_timeout 75s;
        }
        
        # ML service proxy (if separate service)
        location /ml-api {
            proxy_pass http://localhost:5000;

            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_read_timeout 300s;
        }
        
        # Frontend SPA routing
        location / {
            try_files $uri $uri/ /index.html;
            expires 1h;
            add_header Cache-Control "public, immutable";
        }
